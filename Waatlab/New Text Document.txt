from math import gcd

def RSA(p,q,m):
    n=p*q
    t=(p-1)*(q-1)

    e=2
    while gcd(e,t)!=1: e+=1

    d=pow(e,-1,t)

    c=pow(m,e,n)
    print("Encrypted message is",c)

    print("Decrypted message is",pow(c,d,n))

import hashlib
def hash_step(text, algo):
    data = text.encode()
    print("Step 1: Input in bytes:", data.hex())
    if algo == "SHA-1":
        h = hashlib.sha1()
    else:
        h = hashlib.sha512()
    print("Step 2:", algo, "initialized")
    h.update(data)
    print("Step 3: Intermediate Hash:", h.hexdigest())
    print("Step 4: Final Hash:", h.hexdigest())
text = input("Enter text: ")
algo = input("Enter algorithm (SHA-1 / SHA-512): ")
hash_step(text, algo)

def mod_inv(a):
    for i in range(26):
        if (a*i) % 26 == 1:
            return i

def inv_key(k):
    d = (k[0][0]*k[1][1] - k[0][1]*k[1][0]) % 26
    di = mod_inv(d)
    return [[ k[1][1]*di % 26, -k[0][1]*di % 26],
            [-k[1][0]*di % 26,  k[0][0]*di % 26]]

def process(text):
    text = text.replace(" ", "").upper()
    return text + ('X' if len(text) % 2 else '')

def hill(text, k):
    nums = [ord(c)-65 for c in text]
    res = []
    for i in range(0, len(nums), 2):
        a, b = nums[i], nums[i+1]
        res += [(k[0][0]*a + k[0][1]*b) % 26,
                (k[1][0]*a + k[1][1]*b) % 26]
    return ''.join(chr(n+65) for n in res)

key = [[3, 3], [2, 5]]
text = process(input("Enter text: "))

enc = hill(text, key)
dec = hill(enc, inv_key(key))

print("Encrypted:", enc)
print("Decrypted:", dec)
